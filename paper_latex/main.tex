\begin{abstract}
We derive a closed-form expression for the integrated squared Laplacian roughness of a multivariate Gaussian kernel density estimator. The result is a three-term polynomial $P_d(t) = t^2/16 - (d+2)t/4 + d(d+2)/4$ in the scaled pairwise squared distance $t$, enabling non-iterative computation of the roughness functional in any dimension $d$. We use this expression in a one-stage direct plug-in bandwidth selector with a Silverman pilot---the same approach underlying practical implementations of the Sheather--Jones methodology such as R's \texttt{locfit::sjpi}. The resulting scalar bandwidth minimizes the asymptotic mean integrated squared error for isotropic Gaussian KDE without requiring iterative optimization or matrix bandwidth parameterization. We provide scalable computational strategies---subsampling, spatial truncation, and FFT-based convolution---that reduce the $O(n^2 d)$ exact computation to practical running times for large datasets. Empirical evaluation on synthetic mixtures and real datasets demonstrates substantial improvement over rule-of-thumb selectors on multimodal data while maintaining comparable performance on unimodal data. The method is implemented in the open-source Python package \texttt{gsj}, available on PyPI.

\medskip
\noindent\textbf{Keywords:} bandwidth selection, kernel density estimation, plug-in method, multivariate, nonparametric, AMISE, Laplacian roughness
\end{abstract}

\section{Introduction}
\label{sec:intro}

Kernel density estimation (KDE) is a fundamental tool in nonparametric statistics for estimating probability density functions from observed data. Given observations $\bX_1, \ldots, \bX_n \in \R^d$, the Gaussian kernel density estimator with bandwidth matrix $\bm{H}$ is
\begin{equation}
  \hat{f}_{\bm{H}}(\bx) = \frac{1}{n} \sum_{i=1}^n K_{\bm{H}}(\bx - \bX_i),
  \label{eq:kde}
\end{equation}
where $K_{\bm{H}}(\bt) = (2\pi)^{-d/2} |\bm{H}|^{-1/2} \exp(-\frac{1}{2}\bt^T \bm{H}^{-1} \bt)$.

The quality of the estimator depends critically on the bandwidth $\bm{H}$. While sophisticated matrix bandwidth selectors exist \citep{duong2003,duong2005,chacon2010}, they require iterative optimization over $d(d+1)/2$ parameters and are computationally expensive for moderate $d$. At the other extreme, simple rules of thumb \citep{scott1992,silverman1986} assume a Gaussian reference density and can severely oversmooth multimodal data.

In one dimension, the Sheather--Jones (SJ) plug-in selector \citep{sheather1991} provides an effective middle ground: it estimates the roughness of the density's second derivative from the data and plugs this into the AMISE-optimal formula. The core of SJ is a closed-form evaluation of the Gaussian second-derivative roughness integral. However, the full SJ algorithm also includes a multi-stage pilot bandwidth selection procedure that achieves optimal convergence rates \citep{hall1991}. No comparable closed-form roughness expression has been available for $d > 1$.

\textbf{Contribution.} We derive the $d$-dimensional analog of this closed-form roughness expression---a compact polynomial $P_d(t)$ governing the pairwise contribution to the integrated squared Laplacian. This enables a practical one-stage plug-in bandwidth selector for multivariate Gaussian KDE. Specifically, we provide:
\begin{enumerate}
  \item A closed-form expression for $\int [\nabla^2 K_{h_0}(\bx - \bX_i)][\nabla^2 K_{h_0}(\bx - \bX_j)] \, d\bx$ in arbitrary dimension $d$ (Section~\ref{sec:derivation});
  \item Verification that the formula reduces to the 1D Gaussian roughness expression when $d=1$ (Section~\ref{sec:relation});
  \item Scalable computational strategies for large datasets (Section~\ref{sec:computation});
  \item Comprehensive empirical evaluation (Section~\ref{sec:experiments});
  \item An open-source implementation as the \texttt{gsj} Python package (Section~\ref{sec:software}).
\end{enumerate}

\section{Background}
\label{sec:background}

\subsection{AMISE for Isotropic Bandwidth in Whitened Coordinates}
\label{sec:amise}

We consider data that has been pre-whitened: $\bY_i = \hat{\bSigma}^{-1/2}\bX_i$, where $\hat{\bSigma}$ is the sample covariance of the original data $\bX_1, \ldots, \bX_n$. In the whitened space, the KDE uses an isotropic bandwidth $\bm{H} = h^2 \Id$, corresponding to bandwidth matrix $h^2 \hat{\bSigma}$ in the original coordinates.

For this isotropic Gaussian KDE in whitened coordinates, the asymptotic mean integrated squared error is
\begin{equation}
  \text{AMISE}(h) = \frac{R(K)}{nh^d} + \frac{h^4}{4}\Psi,
  \label{eq:amise}
\end{equation}
where $R(K) = (4\pi)^{-d/2}$ is the roughness of the standard $d$-dimensional Gaussian kernel, and
\begin{equation}
  \Psi = \int [\nabla^2 f(\bx)]^2 \, d\bx
  \label{eq:psi_def}
\end{equation}
is the integrated squared Laplacian of the true density $f$ in the whitened space.

\begin{remark}
The use of the Laplacian $\nabla^2 f$ (rather than the full Hessian trace $\tr\{\bSigma \nabla^2 f\}$) as the bias functional is appropriate specifically because we work in whitened coordinates where $\bSigma = \Id$. In original coordinates with general $\bm{H} = h^2 \bSigma$, the bias functional would involve $\tr\{\bSigma \nabla^2 f\}$, which reduces to $\nabla^2 f$ after the whitening transform.
\end{remark}

Minimizing~\eqref{eq:amise} with respect to $h$ yields the AMISE-optimal bandwidth:
\begin{equation}
  h^*_{\text{AMISE}} = \left(\frac{d \cdot R(K)}{n \cdot \Psi}\right)^{1/(d+4)}.
  \label{eq:h_star}
\end{equation}

\subsection{The Plug-In Approach}

Since $\Psi$ depends on the unknown density $f$, the plug-in approach estimates it from the data using a pilot bandwidth $h_0$:
\begin{equation}
  \hat{\Psi}(h_0) = \int \left[\nabla^2 \hat{f}_{h_0}(\bx)\right]^2 d\bx = \frac{1}{n^2} \sum_{i=1}^n \sum_{j=1}^n \int \nabla^2 K_{h_0}(\bx - \bY_i) \cdot \nabla^2 K_{h_0}(\bx - \bY_j) \, d\bx.
  \label{eq:roughness_pairwise}
\end{equation}

We note that $\hat{\Psi}(h_0)$ is an estimate of $R(\nabla^2 \hat{f}_{h_0})$---the roughness of the Laplacian of the \emph{smoothed} KDE---which targets $\Psi = R(\nabla^2 f)$ as $h_0 \to 0$ with appropriate rate conditions.

\subsection{The Original Sheather--Jones Algorithm}
\label{sec:original_sj}

For context, the full Sheather--Jones (1991) algorithm in one dimension involves a \emph{two-stage} pilot selection:
\begin{enumerate}
  \item Estimate $\psi_6 = R(f^{(3)})$ using a normal reference: $\hat{\psi}_6^{NR} = -15/(16\sqrt{\pi})\hat{\sigma}^{-7}$.
  \item Derive a data-driven pilot $g_2$ from $\hat{\psi}_6$: $g_2 = (-6/(\sqrt{2\pi}\hat{\psi}_6 n))^{1/7}$.
  \item Estimate $\psi_4 = R(f'')$ using pilot $g_2$.
  \item Plug into AMISE: $\hat{h} = (R(K)/(n\hat{\psi}_4))^{1/5}$.
\end{enumerate}

This two-stage approach achieves the convergence rate $O(n^{-4/13})$, compared to $O(n^{-5/14})$ for one-stage plug-in methods \citep{hall1991}. The theoretical advantage comes from using a higher-order roughness functional ($\psi_6$) to calibrate the pilot---$\psi_6$ is less sensitive to pilot choice than $\psi_4$, so the normal-reference estimate suffices for the first stage.

\textbf{Our approach} uses a one-stage plug-in with a Silverman pilot (Step 4 only, with $g_2$ replaced by the Silverman bandwidth). This is the same variant implemented in practical software such as R's \texttt{locfit::sjpi} \citep{loader1999}. The one-stage approach sacrifices the optimal convergence rate but gains simplicity and direct generalizability to $d$ dimensions. In practice, the $(d+4)$-th root in the AMISE formula dampens pilot-induced errors: a 20\% error in $\hat{\Psi}$ produces only $\sim$3--4\% error in $\hat{h}$.

\subsection{Existing Multivariate Bandwidth Selectors}

\begin{itemize}
  \item \textbf{Scott's rule} \citep{scott1992}: $h = n^{-1/(d+4)}$. Assumes Gaussian reference density.
  \item \textbf{Silverman's rule} \citep{silverman1986}: $h = (4/(n(d+2)))^{1/(d+4)}$. Similar normal reference.
  \item \textbf{Duong--Hazelton} \citep{duong2003}: Full bandwidth matrix via iterative plug-in. $d(d+1)/2$ parameters.
  \item \textbf{Botev et al.} \citep{botev2010}: Diffusion-based, pilot-free. 1D only (relies on FFT grid).
  \item \textbf{Likelihood cross-validation}: Grid search over $h$ maximizing LOO log-likelihood. Any $d$ but expensive.
\end{itemize}

Our method fills a specific gap: a \emph{non-iterative, closed-form} plug-in selector that works in any $d$, without the complexity of full matrix bandwidth optimization or the cost of cross-validation.

\section{Derivation}
\label{sec:derivation}

All derivations below assume whitened data (covariance $= \Id$) and isotropic Gaussian kernel with bandwidth $h_0$.

\subsection{The Multivariate Kernel Laplacian}

\begin{lemma}
\label{lem:laplacian}
For the isotropic Gaussian kernel $K_h(\bt) = (2\pi h^2)^{-d/2} \exp(-\norm{\bt}^2/(2h^2))$, the Laplacian is
\begin{equation}
  \nabla^2 K_h(\bt) = \frac{1}{h^2} K_h(\bt) \left(\frac{\norm{\bt}^2}{h^2} - d\right).
  \label{eq:laplacian}
\end{equation}
\end{lemma}

\begin{proof}
Each second partial derivative contributes $K_h(\bt)(t_k^2/h^4 - 1/h^2)$ via the product rule. Summing over $k = 1, \ldots, d$ and using $\sum_k t_k^2 = \norm{\bt}^2$ gives the result.
\end{proof}

\subsection{Product-of-Gaussians Lemma}

\begin{lemma}
\label{lem:product}
For any $\bm{a}, \bm{b} \in \R^d$ and bandwidth $h > 0$:
\begin{equation}
  K_h(\bx - \bm{a}) \cdot K_h(\bx - \bm{b}) = C_{\bm{ab}} \cdot \phi_{h/\sqrt{2}}\!\left(\bx - \frac{\bm{a}+\bm{b}}{2}\right),
\end{equation}
where $C_{\bm{ab}} = (4\pi h^2)^{-d/2} \exp(-\norm{\bm{a}-\bm{b}}^2/(4h^2))$ and $\phi_\sigma$ denotes the $\mathcal{N}(\bm{0}, \sigma^2 \Id)$ density.
\end{lemma}

\begin{proof}
Complete the square: $\norm{\bx-\bm{a}}^2 + \norm{\bx-\bm{b}}^2 = 2\norm{\bx - (\bm{a}+\bm{b})/2}^2 + \norm{\bm{a}-\bm{b}}^2/2$.
\end{proof}

\subsection{The Pairwise Integral}

Each term in~\eqref{eq:roughness_pairwise} has the form
\begin{equation}
  I_{ij} = \int \nabla^2 K_{h_0}(\bx - \bY_i) \cdot \nabla^2 K_{h_0}(\bx - \bY_j) \, d\bx.
\end{equation}

Substituting Lemma~\ref{lem:laplacian} and applying Lemma~\ref{lem:product}:
\begin{equation}
  I_{ij} = \frac{C_{ij}}{h_0^4} \cdot \E_{\bZ}\!\left[L_{h_0}(\bZ - \bY_i) \cdot L_{h_0}(\bZ - \bY_j)\right],
  \label{eq:Iij_expectation}
\end{equation}
where $L_h(\bt) = \norm{\bt}^2/h^2 - d$ and $\bZ \sim \mathcal{N}((\bY_i+\bY_j)/2, (h_0^2/2)\Id)$.

\subsection{Computing the Expectation}

\begin{theorem}[Main Result]
\label{thm:main}
Let $r_{ij}^2 = \norm{\bY_i - \bY_j}^2/h_0^2$. The pairwise roughness contribution is
\begin{equation}
  I_{ij} = \frac{1}{(4\pi h_0^2)^{d/2}} \exp\!\left(-\frac{r_{ij}^2}{4}\right) P_d(r_{ij}^2),
  \label{eq:Iij_closed}
\end{equation}
where the dimension-dependent polynomial is
\begin{equation}
  \boxed{P_d(t) = \frac{t^2}{16} - \frac{(d+2)t}{4} + \frac{d(d+2)}{4}.}
  \label{eq:Pd}
\end{equation}
\end{theorem}

\begin{proof}
Define $\bU = \bZ - \bY_i$, $\bV = \bZ - \bY_j = \bU - \bdelta$ where $\bdelta = \bY_j - \bY_i$. Then $\bU \sim \mathcal{N}(\bdelta/2, (h_0^2/2)\Id)$ and:
\begin{align}
  \E[L(\bU) \cdot L(\bV)] &= \frac{\E[\norm{\bU}^2\norm{\bV}^2]}{h_0^4} - \frac{d(\E[\norm{\bU}^2] + \E[\norm{\bV}^2])}{h_0^2} + d^2. \label{eq:expansion}
\end{align}

Using moment formulas for $\bW \sim \mathcal{N}(\bm{0}, \Id)$ (specifically $\E[\norm{\bW}^4] = d(d+2)$) and the relation $\bV = \bU - \bdelta$, we compute each term (Appendix~\ref{app:moments}):
\begin{align}
  \E[\norm{\bU}^2] &= \frac{\norm{\bdelta}^2}{4} + \frac{dh_0^2}{2}, \\
  \E[\norm{\bU}^2\norm{\bV}^2] &= \frac{\norm{\bdelta}^4}{16} + \frac{(d-2)\norm{\bdelta}^2 h_0^2}{4} + \frac{d(d+2)h_0^4}{4}.
\end{align}

Substituting into~\eqref{eq:expansion} and setting $t = \norm{\bdelta}^2/h_0^2 = r_{ij}^2$ yields $P_d(t)$.
\end{proof}

\begin{corollary}
\label{cor:1d}
When $d=1$: $P_1(t) = t^2/16 - 3t/4 + 3/4$.
\end{corollary}

\subsection{The Complete Formula}

The roughness estimate is
\begin{equation}
  \hat{\Psi}(h_0) = \frac{1}{n^2 (4\pi)^{d/2} h_0^{d+4}} \sum_{i=1}^n \sum_{j=1}^n \exp\!\left(-\frac{r_{ij}^2}{4}\right) P_d(r_{ij}^2),
  \label{eq:roughness_final}
\end{equation}
and the plug-in bandwidth is
\begin{equation}
  \hat{h} = \left(\frac{d}{n \cdot \hat{\Psi}(h_0) \cdot (4\pi)^{d/2}}\right)^{1/(d+4)},
  \label{eq:h_plugin}
\end{equation}
with pilot $h_0 = (4/(n(d+2)))^{1/(d+4)}$ (Silverman's rule in whitened coordinates).

\section{Relationship to 1D Implementations}
\label{sec:relation}

\subsection{Reduction to the 1D Case}

\begin{theorem}
\label{thm:1d_reduction}
When $d=1$, equations~\eqref{eq:roughness_final}--\eqref{eq:h_plugin} reduce to the closed-form Gaussian second-derivative roughness calculation underlying the one-stage direct plug-in variant of SJ bandwidth selection.
\end{theorem}

\begin{proof}
Setting $d=1$: $P_1(t) = t^2/16 - 3t/4 + 3/4$; $(4\pi)^{1/2} = 2\sqrt{\pi}$; the exponent becomes $1/(1+4) = 1/5$; and the roughness reduces to $\hat{\Psi} = (1/(n^2 \cdot 2\sqrt{\pi} \cdot h_0^5)) \sum_{i,j} e^{-r_{ij}^2/4} P_1(r_{ij}^2)$. Verified symbolically (SymPy, all 11 component identities) and numerically (relative difference $< 10^{-14}$ across test datasets).
\end{proof}

\subsection{What We Generalize vs.\ What We Do Not}

To be precise about the scope of generalization:

\textbf{Generalized to $d$ dimensions:}
\begin{itemize}
  \item The closed-form Gaussian roughness integral (Theorem~\ref{thm:main})
  \item The AMISE plug-in formula with Silverman pilot
  \item The computational algorithm (exact, subsampled, truncated, FFT)
\end{itemize}

\textbf{Not generalized (and identified as future work):}
\begin{itemize}
  \item The two-stage pilot chain ($\psi_6 \to g_2$), which requires the $d$-dimensional $\psi_6$ roughness functional---a substantially harder calculation involving 6th-order multivariate Gaussian moments
  \item The solve-the-equation (STE) variant, which requires iterative root-finding that does not have a natural multivariate extension
  \item Convergence rate proofs (conjectured, not proven; see Section~\ref{sec:theory})
\end{itemize}

\subsection{Pilot Sensitivity and the \texorpdfstring{$(d+4)$-th}{(d+4)-th} Root}

The one-stage pilot choice is theoretically suboptimal compared to SJ's two-stage approach. However, the $(d+4)$-th root in~\eqref{eq:h_plugin} provides strong dampening of pilot errors:
\begin{equation}
  \frac{\Delta \hat{h}}{\hat{h}} \approx \frac{1}{d+4} \cdot \frac{\Delta \hat{\Psi}}{\hat{\Psi}}.
\end{equation}

For $d=2$: a 30\% error in $\hat{\Psi}$ produces only 5\% error in $\hat{h}$. Empirically, the Silverman pilot gives bandwidth estimates within 1--3\% of the two-stage result on all tested datasets (Section~\ref{sec:experiments}).

\section{Computational Methods}
\label{sec:computation}

The exact computation of~\eqref{eq:roughness_final} requires $O(n^2 d)$ operations: $n^2$ pairwise distances at cost $O(d)$ each, plus $O(n^2)$ polynomial evaluations and exponentials. This is practical for $n \leq 3{,}000$. For larger datasets, we provide four strategies.

\subsection{Random Subsampling}

Sample $m$ random pairs $(i,j)$ uniformly from $\{1,\ldots,n\}^2$ and scale:
\begin{equation}
  \hat{S}_{\text{sub}} = \frac{n^2}{m} \sum_{k=1}^m e^{-r_k^2/4} P_d(r_k^2) + n \cdot P_d(0).
\end{equation}
The second term accounts for diagonal contributions ($r_{ii} = 0$, $P_d(0) = d(d+2)/4$) exactly. Cost: $O(md)$, independent of $n$.

Empirically, $m = 50{,}000$--$80{,}000$ random pairs produces bandwidth estimates within 2\% of the exact computation across all tested scenarios (see Table~\ref{tab:timing}). A formal variance bound is not provided here and is identified as a direction for future theoretical work.

\subsection{Spatial Truncation via KD-Tree}

The summand $e^{-r^2/4} P_d(r^2)$ decays rapidly for large $r$. The maximum magnitude at distance $r$ is bounded by $|P_d(r^2)| e^{-r^2/4}$. Since $P_d$ grows quadratically while the exponential decays, we compute the effective cutoff:

For $d \leq 10$ and cutoff $c = 8$ (i.e., $\norm{\bY_i - \bY_j} > 8h_0$): $|P_d(64)| \cdot e^{-16} < 2 \times 10^{-5}$ for all $d \leq 10$.

Using a KD-tree to enumerate only pairs within radius $R_{\text{cut}} = 8h_0$ reduces cost to $O(n \cdot k)$ where $k$ is the average neighborhood size.

\subsection{FFT-Based Binning (for \texorpdfstring{$d \leq 3$}{d <= 3})}

Discretize data onto an $M^d$ grid, compute the kernel on the same grid, and evaluate the roughness sum via FFT convolution. Cost: $O(nd + M^d \log M)$, independent of $n$. This is analogous to how R's \texttt{bw.SJ} achieves practical speed via linear binning on $M \approx 401$ grid points.

\subsection{GPU Acceleration}

The pairwise computation is embarrassingly parallel. Using PyTorch's \texttt{torch.cdist} or CuPy's broadcasting on GPU hardware yields 50--75$\times$ speedups over single-threaded NumPy.

\begin{algorithm}[t]
\caption{Plug-In Bandwidth Selection}
\label{alg:adaptive}
\begin{algorithmic}[1]
\Require Data $\bX_1, \ldots, \bX_n \in \R^d$
\State Whiten: $\bY_i \gets \hat{\bSigma}^{-1/2}\bX_i$ \Comment{Covariance $\to \Id$}
\State Pilot: $h_0 \gets (4/(n(d+2)))^{1/(d+4)}$ \Comment{Silverman in whitened space}
\If{$n \leq 3000$}
  \State Exact pairwise computation~\eqref{eq:roughness_final}
\ElsIf{$d \leq 3$ and $n > 10000$}
  \State FFT-based binning
\Else
  \State Subsampling with $m = 80{,}000$ pairs
\EndIf
\State \Return $\hat{h}$ via~\eqref{eq:h_plugin}
\Comment{Bandwidth matrix in original space: $h^2 \hat{\bSigma}$}
\end{algorithmic}
\end{algorithm}

\section{Theoretical Properties}
\label{sec:theory}

\subsection{Finite-Sample Exactness}

Theorem~\ref{thm:main} provides the \emph{exact} value of $R(\nabla^2 \hat{f}_{h_0})$ for any finite sample. This is a deterministic identity conditional on the data---not an approximation. The polynomial $P_d(t)$ is exact for the Gaussian kernel.

\subsection{Asymptotic Behavior}

The plug-in estimator~\eqref{eq:h_plugin} targets $h^*_{\text{AMISE}}$ as $n \to \infty$. A formal consistency proof requires establishing that $\hat{\Psi}(h_0) \to \Psi$ under the Silverman pilot $h_0 = O(n^{-1/(d+4)})$. At this rate, $nh_0^{d+4} = 4/(d+2)$---a constant rather than a diverging quantity.

Standard plug-in consistency arguments require $nh_0^{d+4} \to \infty$, which is not satisfied by the Silverman pilot. However, the estimator $\hat{\Psi}$ is a U-statistic-type quantity whose asymptotic behavior at the critical rate $h_0 \sim n^{-1/(d+4)}$ may be analyzed via different techniques (degenerate U-statistic theory). A formal treatment is beyond the scope of this paper. Empirical evidence strongly suggests consistency: the bandwidth stabilizes as $n$ increases across all tested distributions (Section~\ref{sec:experiments}).

\subsection{Convergence Rate (Conjecture)}

Based on analogy with the 1D one-stage plug-in rate $O(n^{-5/14})$ \citep{hall1991}, we conjecture:
\begin{equation}
  \frac{|\hat{h} - h^*_{\text{AMISE}}|}{h^*_{\text{AMISE}}} = O_p\!\left(n^{-5/(3d+14)}\right).
\end{equation}
Numerical evidence supports this conjecture (see supplementary materials). A formal proof requires extending the Hall--Marron stochastic expansion to the multivariate setting and is left for future work.

\subsection{Comparison of Convergence Rates}

from gsj import bandwidth
import numpy as np

X = np.random.randn(2000, 5)

The package supports NumPy, PyTorch, and CuPy backends, with automatic method selection based on dataset size.

\section{Discussion}
\label{sec:discussion}

\subsection{When to Use This Method}

The method is most beneficial for multimodal data (clusters, mixtures) in $d = 2$--$10$ with moderate sample sizes ($n = 100$--$100{,}000$). On unimodal Gaussian data, it provides 5--15\% worse ISE than the optimal Silverman rule---an acceptable trade-off given gains of 65--90\% ISE reduction on structured data.

\subsection{Limitations}

\begin{enumerate}
  \item \textbf{Scalar bandwidth}: Finds the best isotropic bandwidth. Highly anisotropic data may benefit from diagonal or full matrix selectors.
  \item \textbf{Curse of dimensionality}: For $d > 15$--$20$, KDE itself degrades. Dimensionality reduction (PCA) is required.
  \item \textbf{Pilot dependence}: Results depend on Silverman pilot, though weakly via the $(d+4)$-th root.
  \item \textbf{Gaussian kernel only}: The closed form is specific to the Gaussian kernel. Other kernels require numerical integration.
\end{enumerate}

\subsection{Future Work}

\begin{itemize}
  \item \textbf{Two-stage pilot in $d$ dimensions}: Deriving the closed-form $\psi_6$ functional (6th-order Gaussian moments) would enable a two-stage approach with optimal convergence rate.
  \item \textbf{Formal consistency proof}: Via U-statistic theory at the critical rate $h_0 \sim n^{-1/(d+4)}$.
  \item \textbf{Diagonal bandwidth}: Per-axis roughness functionals for a diagonal bandwidth matrix.
  \item \textbf{STE variant}: An iterative self-consistent version for $d > 1$.
\end{itemize}

\section{Moment Calculations}
\label{app:moments}

\subsection{Fourth Moment of the Squared Norm}

For $\bW \sim \mathcal{N}(\bm{0}, \Id)$:
\begin{align}
  \E[\norm{\bW}^4] &= \sum_k \E[W_k^4] + 2\sum_{k<l} \E[W_k^2]\E[W_l^2] = 3d + d(d-1) = d(d+2).
\end{align}

\subsection{Cross Term}

For $\bU = \bmu + \sigma\bW$:
\begin{align}
  \E[\norm{\bU}^2(\bU^T\bdelta)] &= (\bmu^T\bdelta)\left[\norm{\bmu}^2 + (d+2)\sigma^2\right].
\end{align}

Derivation: expand $\norm{\bU}^2 = \norm{\bmu}^2 + 2\sigma\bmu^T\bW + \sigma^2\norm{\bW}^2$ and $\bU^T\bdelta = \bmu^T\bdelta + \sigma\bW^T\bdelta$. Cross terms with odd powers of $\bW$ vanish. The non-vanishing terms are: $\norm{\bmu}^2(\bmu^T\bdelta)$, $2\sigma^2(\bmu^T I_d \bdelta) = 2\sigma^2(\bmu^T\bdelta)$ from $\E[(\bmu^T\bW)(\bW^T\bdelta)]$, and $\sigma^2 d (\bmu^T\bdelta)$ from $\E[\norm{\bW}^2](\bmu^T\bdelta)$.

\subsection{Assembling the Fourth Mixed Moment}

Using $\bV = \bU - \bdelta$:
\begin{align}
  \E[\norm{\bU}^2\norm{\bV}^2] &= \E[\norm{\bU}^4] - 2\E[\norm{\bU}^2(\bU^T\bdelta)] + \norm{\bdelta}^2\E[\norm{\bU}^2].
\end{align}

Substituting $\norm{\bmu}^2 = \norm{\bdelta}^2/4$ and $\sigma^2 = h_0^2/2$ yields:
\[
  \E[\norm{\bU}^2\norm{\bV}^2] = \frac{\norm{\bdelta}^4}{16} + \frac{(d-2)\norm{\bdelta}^2 h_0^2}{4} + \frac{d(d+2)h_0^4}{4}.
\]
