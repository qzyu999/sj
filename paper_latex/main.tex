\begin{abstract}
We generalize the Sheather--Jones (1991) bandwidth selection method to $d$~dimensions. The generalization rests on two closed-form polynomials---$P_d$ for estimating the Laplacian roughness~$\Psi_4$ and $R_d$ for the gradient-of-Laplacian roughness~$\Psi_6$---which enable a complete two-stage plug-in algorithm with an adaptive pilot bandwidth derived from bias cancellation, exactly mirroring the original one-dimensional construction. The method reduces to classical Sheather--Jones when $d=1$ and is implemented in the open-source Python package \texttt{gsj}. We demonstrate its practical value on intraday volume profile analysis, where the data-driven bandwidth consistently identifies 1--2~additional support/resistance levels compared to normal-reference rules.

\medskip
\noindent\textbf{Keywords:} bandwidth selection, kernel density estimation, plug-in method, multivariate, Sheather--Jones, volume profile
\end{abstract}

% ============================================================================
\section{Introduction}
\label{sec:intro}

The Sheather--Jones (SJ) bandwidth selector \citep{sheather1991} has been the default method for univariate kernel density estimation for over three decades. Its success rests on three properties: (i)~a closed-form roughness estimator that is always positive, (ii)~an adaptive pilot bandwidth derived from bias cancellation, and (iii)~superior convergence rate $O_p(n^{-5/14})$ compared to cross-validation or normal-reference methods.

Despite this, no complete multivariate analog has been available. Practitioners working with $d$-dimensional data---embeddings, financial time series, sensor arrays---default to Silverman's normal-reference rule \citep{silverman1986} or computationally expensive cross-validation. Matrix bandwidth selectors \citep{chacon2010} require iterating over $d(d+1)/2$ parameters.

We provide the missing piece: a closed-form polynomial expression for multivariate Gaussian KDE roughness that enables the entire SJ architecture to be replicated in any dimension. The contribution is:
\begin{enumerate}
  \item Two polynomials ($P_d$ and $R_d$) that compute the roughness functionals $\Psi_4$ and $\Psi_6$ from pairwise distances (Theorems~\ref{thm:Pd} and~\ref{thm:Rd});
  \item A complete two-stage plug-in algorithm with data-driven pilot (Algorithm~\ref{alg:full});
  \item Demonstration on real intraday market data where the method detects finer price structure than normal-reference rules.
\end{enumerate}

\section{Main Results}
\label{sec:results}

We work in whitened coordinates $\bY_i = \hat\bSigma^{-1/2}(\bX_i - \bar\bX)$ with isotropic bandwidth~$h$. The AMISE of the Gaussian KDE is
\begin{equation}
  \text{AMISE}(h) = \frac{(4\pi)^{-d/2}}{nh^d} + \frac{h^4}{4}\,\Psi_4,
  \qquad \Psi_4 = \int [\nabla^2 f(\bx)]^2\,d\bx,
  \label{eq:amise}
\end{equation}
minimized at $h^* = (d\,(4\pi)^{-d/2}/(n\Psi_4))^{1/(d+4)}$. To estimate~$\Psi_4$, we integrate products of Laplacian kernels over all data pairs.

\begin{theorem}[Roughness polynomial $P_d$]
\label{thm:Pd}
Let $r_{ij}^2 = \norm{\bY_i - \bY_j}^2/h_0^2$. The pairwise contribution to~$\hat\Psi_4$ is
\[
  I_{ij} = \frac{e^{-r_{ij}^2/4}}{(4\pi h_0^2)^{d/2}} \cdot P_d(r_{ij}^2),
  \qquad
  P_d(t) = \frac{t^2}{16} - \frac{(d+2)\,t}{4} + \frac{d(d+2)}{4}.
\]
\end{theorem}

\begin{theorem}[Gradient-of-Laplacian polynomial $R_d$]
\label{thm:Rd}
The next-order functional $\Psi_6 = \int \norm{\nabla(\nabla^2 f)}^2\,d\bx$ is estimated with
\[
  R_d(t) = -\frac{t^3}{64} + \frac{3(d+4)}{32}\,t^2 - \frac{3(d+2)(d+4)}{16}\,t + \frac{d(d+2)(d+4)}{8}.
\]
\end{theorem}

Both polynomials are positive at $t=0$ (ensuring positivity of the estimator) and reduce to the classical Hermite-function formulas at $d=1$ (Corollary~\ref{cor:d1} in the Supplement).

\begin{remark}
$P_d$ has degree~2 and $R_d$ has degree~3. Each additional derivative order adds one degree and one factor of $(d+2k)$ to the constant term at $t=0$.
\end{remark}

\section{The Algorithm}
\label{sec:algorithm}

The full two-stage method mirrors SJ's original structure:

\begin{algorithm}[t]
\caption{Multivariate Sheather--Jones Bandwidth Selection}
\label{alg:full}
\begin{algorithmic}[1]
\Require Data $\bX_1,\ldots,\bX_n \in \R^d$
\State \textbf{Whiten:} $\bY_i \gets \hat\bSigma^{-1/2}(\bX_i - \bar\bX)$
\State \textbf{Scale:} $\hat\sigma \gets \text{median}(\norm{\bY_i}) \,/\, \chi_d^{\text{med}}$ \Comment{robust scale via chi distribution median}
\State \textbf{NR pilot:} $g \gets \hat\sigma\,(2/(n(d{+}4)))^{1/(d+6)}$ \Comment{bias-cancellation bandwidth}
\State \textbf{Estimate $\Psi_6$:} $\hat\Psi_6 \gets \frac{1}{n^2(4\pi)^{d/2}g^{d+6}}\sum_{i,j}e^{-r_{ij}^2/4}\,R_d(r_{ij}^2)$
\State \textbf{Adaptive pilot:} $g_1 \gets \bigl(d(d{+}2)\,/\,(4n(4\pi)^{d/2}\hat\Psi_6)\bigr)^{1/(d+6)}$
\State \textbf{Estimate $\Psi_4$:} $\hat\Psi_4 \gets \frac{1}{n^2(4\pi)^{d/2}g_1^{d+4}}\sum_{i,j}e^{-r_{ij}^2/4}\,P_d(r_{ij}^2)$
\State \textbf{Bandwidth:} $\hat h \gets \bigl(d\,/\,(n\hat\Psi_4(4\pi)^{d/2})\bigr)^{1/(d+4)}$
\State \Return $\hat h$; full bandwidth matrix $\bm{H} = \hat h^2\,\hat\bSigma$
\end{algorithmic}
\end{algorithm}

\textbf{Simplified variants.} A one-stage DPI uses Silverman's pilot directly ($g_1 = (4/(n(d{+}2)))^{1/(d+4)}$), skipping Steps~3--5. A solve-the-equation (STE) variant couples the pilot with~$h$ via $g_1(h) \propto h^{(d+4)/(d+6)}$ and solves the resulting fixed-point equation numerically. All three are available in the \texttt{gsj} package.

\section{Computation}
\label{sec:computation}

The exact computation is $O(n^2 d)$. For large~$n$:
\begin{itemize}
  \item \textbf{Subsampling:} Sample $m$ random pairs (default $m{=}80{,}000$); scale by~$n^2/m$. Bandwidth error: ${\sim}\,1$--$2\%$ (roughness error $5$--$11\%$ dampened by the $(d{+}4)$-th root).
  \item \textbf{Spatial truncation:} KD-tree enumerates pairs within $8h_0$. Error $<2.4\times10^{-5}$ for $d\leq 10$.
  \item \textbf{FFT ($d\leq 3$):} Grid binning + convolution, $O(nd + M^d\log M)$.
\end{itemize}
With subsampling, all algorithm variants (one-stage, two-stage, STE) cost ${\sim}\,100$--$200\,$ms regardless of~$n$.

\section{Application: Volume Profile Level Detection}
\label{sec:volume_profile}

A direct application where bandwidth choice has immediate practical consequences is \emph{volume profile analysis} in financial markets. Traders identify support/resistance levels as modes of the volume-weighted price density within a trading session. The bandwidth determines how many levels are resolved.

We computed rolling 60-bar volume-weighted KDEs on 1-minute data for five US equities (NVDA, TSLA, AAPL, SPY, AMD) over five trading days. At each rolling window, we detected density peaks (modes) using both Silverman's rule and our method.

Across all stocks, our method identifies 1--2 additional levels that visual inspection confirms correspond to genuine price consolidation zones. The data-driven bandwidth is 39--58\% of Silverman's---meaning Silverman oversmooths by a factor of~2 on this class of multimodal data.

\section{Simulation Study}
\label{sec:simulations}

On $d{=}2$ Gaussian data with $n{=}500$, Silverman's rule outperforms our method on ISE by ${\sim}10\%$ (as expected---the normal-reference assumption is exact). On tight 5-component mixtures ($\sigma{=}0.6$, $n{=}500$), our two-stage method achieves lower ISE than Silverman ($p{=}0.006$, paired $t$-test).

The regime map: our method improves on Silverman when (i) the density is multimodal with fine structure and (ii)~$n$ is large enough that the AMISE approximation is accurate ($n \gtrsim 1000$ in $d{=}2$). For unimodal data or small~$n$, Silverman's implicit oversmoothing accidentally compensates for the finite-sample AMISE-to-MISE gap.

This matches the behavior of the 1D SJ method: it becomes advantageous on multimodal data at moderate~$n$ \citep{jones1996brief}.

\section{Discussion}
\label{sec:discussion}

\textbf{Relationship to SJ (1991).} Our method is a direct generalization: the roughness formula, bias-cancellation pilot, and STE structure all reduce to the original at $d{=}1$. The key enabling result is the polynomial~$P_d$, which makes multivariate roughness evaluation a single vectorized summation rather than an iterative optimization.

\textbf{When to use.} Multimodal data in $d=1$--$15$; applications where density peaks are the signal (mode detection, anomaly scoring, volume profile); large~$n$ where cross-validation is too slow.

\textbf{Limitations.} (1)~Isotropic bandwidth only---strongly anisotropic data needs diagonal or full-matrix selectors. (2)~KDE itself degrades for $d>15$--$20$ (curse of dimensionality). (3)~At small~$n$, the AMISE-to-MISE gap means Silverman can win on ISE for near-normal data.

\textbf{Software.} \texttt{pip install gsj}; source at \url{https://github.com/qzyu999/gsj}.

\section{Derivation of $P_d$}
\label{sec:supp_Pd}

\begin{proof}[Proof of Theorem~\ref{thm:Pd}]
The pairwise integral is
\[
I_{ij} = \int [\nabla^2 K_{h_0}(\bx - \bY_i)][\nabla^2 K_{h_0}(\bx - \bY_j)]\,d\bx.
\]
Using the Gaussian product lemma, $K_h(\bx{-}\bm{a})\,K_h(\bx{-}\bm{b}) = C_{\bm{ab}}\,\phi_{h/\sqrt2}(\bx - \bm{m})$ where $\bm{m} = (\bm{a}{+}\bm{b})/2$ and $C_{\bm{ab}} = (4\pi h^2)^{-d/2}e^{-\norm{\bm{a}-\bm{b}}^2/(4h^2)}$, the integral becomes an expectation with respect to $\bW \sim \mathcal{N}(\bm{0}, \Id)$:
\[
I_{ij} = \frac{C_{\bm{ab}}}{h_0^4}\,\E\bigl[L(\bm{U})\,L(\bm{V})\bigr],
\]
where $\bm{U} = \bW/\sqrt2 - \bm{\delta}/(2h_0)$, $\bm{V} = \bW/\sqrt2 + \bm{\delta}/(2h_0)$, $\bm{\delta} = \bY_i - \bY_j$, and $L(\bm{t}) = \norm{\bm{t}}^2 - d$ is the Laplacian eigenfunction.

Expanding with $S = \norm{\bW}^2 \sim \chi^2_d$ and $r^2 = \norm{\bm{\delta}}^2/h_0^2$:
\begin{align*}
\E[L(\bm{U})\,L(\bm{V})] &= \E\!\left[\left(\tfrac{S}{2} - \tfrac{r^2}{4} - d\right)\!\left(\tfrac{S}{2} + \tfrac{r^2}{4} - d\right) + \text{cross terms}\right].
\end{align*}
Using $\E[S] = d$, $\E[S^2] = d(d{+}2)$, and cancellation of odd-moment cross terms, we obtain after algebra (detailed in Supplement~S1.1):
\[
\E[L(\bm{U})L(\bm{V})] = \frac{r^4}{16} - \frac{(d{+}2)r^2}{4} + \frac{d(d{+}2)}{4} = P_d(r^2).
\]
\end{proof}

\subsection{Moment details}
\label{sec:supp_moments}
For $\bW \sim \mathcal{N}(\bm 0, \Id)$: $\E[\norm{\bW}^4] = d(d+2)$. For $\bm{U} = \sigma\bW + \bm\mu$:
\begin{align*}
\E[\norm{\bm{U}}^2] &= d\sigma^2 + \norm{\bm\mu}^2,\\
\E[\norm{\bm{U}}^4] &= d(d{+}2)\sigma^4 + 2(d{+}2)\sigma^2\norm{\bm\mu}^2 + \norm{\bm\mu}^4.
\end{align*}
These were verified symbolically (SymPy) and numerically (relative difference $<10^{-14}$ between $d$-D formula at $d{=}1$ and dedicated 1D implementation).

\section{Derivation of $R_d$}
\label{sec:supp_Rd}

\begin{proof}[Proof of Theorem~\ref{thm:Rd}]
The gradient of the Laplacian of the Gaussian kernel is:
\[
\nabla(\nabla^2 K_h(\bm{t})) = K_h(\bm{t})\,\frac{\bm{t}}{h^4}\,\bigl[(d{+}2) - \norm{\bm{t}}^2/h^2\bigr].
\]
The pairwise dot-product integral becomes:
\[
J_{ij} = \frac{C_{\bm{ab}}}{h_0^6}\,\E\bigl[(\bm{u}\!\cdot\!\bm{v})\,((d{+}2){-}\norm{\bm{u}}^2)\,((d{+}2){-}\norm{\bm{v}}^2)\bigr],
\]
where $\bm{u} = \bW/\sqrt2 - \bm\mu$, $\bm{v} = \bW/\sqrt2 + \bm\mu$, $\norm{\bm\mu}^2 = r^2/4$.

Key intermediate quantities (with $S = \norm{\bW}^2$, $Z = \bW\!\cdot\!\hat{\bm\mu} \sim N(0,1)$):
\begin{align*}
\bm{u}\!\cdot\!\bm{v} &= S/2 - r^2/4,\\
(d{+}2) - \norm{\bm{u}}^2 &= (d{+}2) - S/2 + Zr/\sqrt2 - r^2/4,\\
(d{+}2) - \norm{\bm{v}}^2 &= (d{+}2) - S/2 - Zr/\sqrt2 - r^2/4.
\end{align*}
Computing $\E[(\bm{u}\!\cdot\!\bm{v})((d{+}2){-}\norm{\bm{u}}^2)((d{+}2){-}\norm{\bm{v}}^2)]$ using $S$ and $Z^2$ moments (noting $S = Z^2 + S'$ with $S' \sim \chi^2_{d-1}$ independent of~$Z$) yields after expansion:
\[
R_d(r^2) = -\frac{r^6}{64} + \frac{3(d{+}4)}{32}\,r^4 - \frac{3(d{+}2)(d{+}4)}{16}\,r^2 + \frac{d(d{+}2)(d{+}4)}{8}.
\]
Verified by Monte Carlo simulation (relative error $<10^{-3}$ for $d = 1,2,3,5,10$).
\end{proof}

\section{Bias Cancellation and Pilot Bandwidth}
\label{sec:supp_pilot}

The expectation of our estimator (for $i \neq j$) is:
\[
\E[\hat\Psi_4(h_0)] = \Psi_4 - h_0^2\,\Psi_6 + \frac{P_d(0)}{n\,(4\pi)^{d/2}\,h_0^{d+4}} + O(h_0^4).
\]
The first correction (smoothing bias) is negative; the second (diagonal) is positive. Setting them equal:
\[
h_0^2\,\Psi_6 = \frac{d(d{+}2)}{4n(4\pi)^{d/2}\,h_0^{d+4}}
\quad\Rightarrow\quad
h_0^{d+6} = \frac{d(d{+}2)}{4n(4\pi)^{d/2}\,\Psi_6}.
\]
Using the normal reference $\Psi_6^{NR}(\sigma) = d(d{+}2)(d{+}4)/(8(4\pi)^{d/2}\sigma^{d+6})$ gives
\[
g = \hat\sigma\,\Bigl(\frac{2}{n(d{+}4)}\Bigr)^{\!1/(d+6)}.
\]
At $d{=}1$: $g \propto n^{-1/7}$, matching the pilot rate in~\citet{sheather1991}.

% ============================================================================
\section{Normal References}
\label{sec:supp_NR}

For $f = \mathcal{N}(\bm 0, \sigma^2\Id)$:
\begin{align*}
\Psi_4^{NR}(\sigma) &= \frac{d(d+2)}{4(4\pi)^{d/2}\,\sigma^{d+4}}, \\
\Psi_6^{NR}(\sigma) &= \frac{d(d+2)(d+4)}{8(4\pi)^{d/2}\,\sigma^{d+6}}.
\end{align*}
At $d{=}1$, $\sigma{=}1$: $\Psi_4^{NR} = 3/(8\sqrt\pi)$ and $\Psi_6^{NR} = 15/(16\sqrt\pi)$, matching the known values $R(\phi'')$ and $R(\phi''')$.

\section{Reduction to Classical SJ at $d=1$}
\label{sec:supp_d1}

\begin{corollary}
\label{cor:d1}
At $d=1$: $P_1(t) = t^2/16 - 3t/4 + 3/4$. This satisfies
\[
\frac{e^{-z^2/4}}{\sqrt{4\pi}\,h_0^5}\,P_1(z^2) = \frac{1}{h_0^5}\,\phi^{(4)}\!\bigl(z/\sqrt2\bigr),
\]
where $\phi^{(4)}(u) = (u^4-6u^2+3)\phi(u)$ and $z = (X_i{-}X_j)/h_0$. Our estimator at bandwidth~$h_0$ equals SJ's $\hat S_D(\alpha)$ at $\alpha = h_0\sqrt2$.

Similarly, $R_1(t) = -t^3/64 + 15t^2/32 - 45t/16 + 15/8$ corresponds to the $\phi^{(6)}$ formula in the integral-of-products framework.
\end{corollary}

\section{Convergence Rate}
\label{sec:supp_rate}

\begin{proposition}[Conjectured]
Under standard smoothness conditions on~$f$, the two-stage estimator satisfies
\[
\frac{|\hat h - h^*|}{h^*} = O_p\!\left(n^{-5/(3d+14)}\right).
\]
\end{proposition}
At $d=1$ this gives $O_p(n^{-5/17})$, which is between SJ's $O_p(n^{-5/14})$ and Park--Marron's $O_p(n^{-4/13})$. A formal proof via degenerate U-statistic theory for the integral-of-products estimator is the subject of ongoing work.
